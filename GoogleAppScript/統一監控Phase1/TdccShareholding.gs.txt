/**
 * ============================================================
 * 每週預建集保「股權分散表」空檔 (給 SA fetch_tdcc.py 覆蓋用)
 * ============================================================
 * 跟 CreatePlaceholders.gs 同一套路:
 *   Service Account 沒有 Drive storage quota,只能「覆蓋既存檔」不能「建立新檔」。
 *   GAS 用你個人帳號跑、有配額,先把空檔建好,GitHub Actions 上的
 *   scripts/fetch_tdcc.py --drive-api 就能順利覆蓋內容。
 *
 * ⚠️ 檔名裡的資料日期**不能用「上一個週五」硬算**:
 *   遇到假日集保會順延/提前 (實際看過 20260709(四)、20260618、20261023…),
 *   所以先去 TDCC 查詢頁 (55KB,很小) 讀第一個 <option> 拿到真正的最新資料日期。
 *
 * 觸發器設法 (Apps Script → ⏰ Triggers → Add Trigger):
 *   函式: runWeeklyTdccPlaceholder
 *   事件來源: 時間驅動 → 週計時器 → 每週五 → 晚上 7 點到 8 點
 *   (要排在 GitHub Actions tdcc-shareholding.yml 的 20:20 之前)
 *
 * 檔名格式與 scripts/fetch_tdcc.py 一致: TDCC_OD_1-5_YYYYMMDD.csv
 * ============================================================
 */

// Drive「股權分散表」資料夾 (Telegram Bot/股權分散表)
// Config.gs 有設 CFG.DRIVE.TDCC_SHAREHOLDING 就優先用那個
var TDCC_FOLDER_ID_FALLBACK = '133xYbjvZXpj7cFLMWIjzkRfYBnipe1VE';
var TDCC_OD_URL = 'https://opendata.tdcc.com.tw/getOD.ashx?id=1-5';
var TDCC_QRY_URL = 'https://www.tdcc.com.tw/portal/zh/smWeb/qryStock';
var TDCC_MIN_BYTES = 500000;  // 全市場 ~4000 檔 × 17 級距,正常 2MB 以上

function runWeeklyTdccPlaceholder() {
  try {
    var r = createTdccPlaceholder();
    Logger.log(JSON.stringify(r));
    return r;
  } catch (e) {
    if (typeof sendTelegramAlert === 'function') {
      sendTelegramAlert('❌ 集保股權分散表空檔預建失敗: ' + e.message);
    }
    throw e;
  }
}

/** 主邏輯:查最新資料日期 → 在 Drive 建同名空檔 (已存在則跳過) */
function createTdccPlaceholder() {
  var folderId = _tdccFolderId();
  var dateStr = getTdccLatestDate();
  var name = 'TDCC_OD_1-5_' + dateStr + '.csv';

  var folder = DriveApp.getFolderById(folderId);
  if (folder.getFilesByName(name).hasNext()) {
    Logger.log('  ⏭ skip (已存在): ' + name);
    return { date: dateStr, name: name, action: 'skipped' };
  }
  folder.createFile(name, '', 'text/csv');
  Logger.log('  ✅ created: ' + name);
  return { date: dateStr, name: name, action: 'created' };
}

/** 從 TDCC 查詢頁的日期下拉選單讀最新資料日期 (第一個 option) */
function getTdccLatestDate() {
  var resp = UrlFetchApp.fetch(TDCC_QRY_URL, {
    muteHttpExceptions: true,
    followRedirects: true,
    validateHttpsCertificates: false,  // TDCC 憑證缺 Subject Key Identifier
    headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)' }
  });
  if (resp.getResponseCode() !== 200) {
    throw new Error('查詢頁 HTTP ' + resp.getResponseCode());
  }
  var m = resp.getContentText('UTF-8').match(/<option value="(\d{8})"/);
  if (!m) throw new Error('查詢頁讀不到資料日期 option');
  return m[1];
}

// ─────────────────────────────────────────────────────────────
// 備援:GAS 自己抓 CSV 存檔 (SA 那條掛掉時手動跑,或另外設觸發器)
// 正常流程用不到 — 內容由 GitHub Actions 的 SA 覆蓋
// ─────────────────────────────────────────────────────────────
function fetchTdccShareholdingToDrive() {
  var resp = UrlFetchApp.fetch(TDCC_OD_URL, {
    muteHttpExceptions: true,
    followRedirects: true,
    validateHttpsCertificates: false,
    headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)' }
  });
  if (resp.getResponseCode() !== 200) throw new Error('HTTP ' + resp.getResponseCode());

  var text = resp.getContentText('UTF-8');
  if (text.length < TDCC_MIN_BYTES) throw new Error('內容過短 len=' + text.length);
  if (text.indexOf('證券代號') < 0) throw new Error('缺欄位「證券代號」');

  // 資料日期:第 2 列 (第 1 列是表頭) 的第 1 欄,整份都同一天
  var firstRow = text.split('\n')[1] || '';
  var dateStr = (firstRow.split(',')[0] || '').trim();
  if (!/^\d{8}$/.test(dateStr)) throw new Error('取不到資料日期: ' + firstRow.slice(0, 60));

  var name = 'TDCC_OD_1-5_' + dateStr + '.csv';
  var folder = DriveApp.getFolderById(_tdccFolderId());
  var it = folder.getFilesByName(name);
  var action;
  if (it.hasNext()) { it.next().setContent(text); action = 'overwritten'; }
  else { folder.createFile(name, text, 'text/csv'); action = 'created'; }

  Logger.log('✅ 集保股權分散表 ' + dateStr + ' ' + action);
  return { date: dateStr, name: name, action: action, bytes: text.length };
}

function _tdccFolderId() {
  return (typeof CFG !== 'undefined' && CFG.DRIVE && CFG.DRIVE.TDCC_SHAREHOLDING)
    ? CFG.DRIVE.TDCC_SHAREHOLDING : TDCC_FOLDER_ID_FALLBACK;
}

  return r;
}
